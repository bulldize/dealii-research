cmake_minimum_required(VERSION 3.20)
find_package(deal.II 9.7 REQUIRED HINTS ${DEAL_II_DIR} ../Dealii/install-local)
deal_ii_initialize_cached_variables()
project(dealii_research LANGUAGES CXX)
add_library(research src/solver.cc)
target_include_directories(research PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
deal_ii_setup_target(research)
add_executable(giesekus apps/giesekus.cc)
target_link_libraries(giesekus research)
deal_ii_setup_target(giesekus)
add_executable(kernel_tests tests/kernels.cc)
target_include_directories(kernel_tests PRIVATE include)
deal_ii_setup_target(kernel_tests)
enable_testing()
add_test(NAME kernels COMMAND kernel_tests)
set_tests_properties(kernels PROPERTIES ENVIRONMENT "DYLD_LIBRARY_PATH=${DEAL_II_PATH}/lib")
